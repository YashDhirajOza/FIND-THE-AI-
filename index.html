find_the_ai_engine.py (Full Multi-Engine Ensemble Implementation) 
import torch
import numpy as np
from transformers import AutoModelForCausalLM, AutoTokenizer, AutoModelForSequenceClassification

class FindTheAIConsensusEngine:
    def __init__(self):
        # 1. Zero-Shot Observer & Performer Models (Binoculars)
        self.obs_tokenizer = AutoTokenizer.from_pretrained("gpt2-medium")
        self.obs_model = AutoModelForCausalLM.from_pretrained("gpt2-medium").eval()
        
        # 2. Neural Disentangled Attention Classifier
        self.cls_tokenizer = AutoTokenizer.from_pretrained("microsoft/deberta-v3-large")
        self.cls_model = AutoModelForSequenceClassification.from_pretrained("microsoft/deberta-v3-large").eval()

    def extract_stylometrics(self, text: str, sentences: list) -> dict:
        lens = [len(s.split()) for s in sentences if len(s.strip()) > 0]
        mean_len = np.mean(lens) if lens else 0.0
        burstiness = float(np.std(lens) / (mean_len + 1e-8)) if lens else 0.0
        words = text.lower().split()
        ttr = len(set(words)) / (len(words) + 1e-8)
        return {"burstiness": burstiness, "ttr": ttr, "mean_len": mean_len}

    def compute_curvature(self, text: str) -> float:
        inputs = self.obs_tokenizer(text, return_tensors="pt")
        with torch.no_grad():
            loss = self.obs_model(**inputs, labels=inputs["input_ids"]).loss
        return float(torch.exp(loss).item())

    def evaluate(self, text: str) -> dict:
        words = text.split()
        # Elena's Safeguard: Abstention check on low sample sizes
        if len(words) < 50:
            return {"verdict": "ABSTAINED", "reason": "Insufficient sample size (< 50 words)"}

        sentences = [s.strip() for s in text.split(".") if len(s.strip()) > 0]
        stylo = self.extract_stylometrics(text, sentences)
        ppl = self.compute_curvature(text)
        
        # Inter-Engine Consensus Calculation
        scores = [stylo["burstiness"], ppl / 100.0]
        consensus = (0.35 * (1.0 - stylo["burstiness"]) + 0.35 * (1.0 / (1.0 + np.exp((ppl - 45)/10)))) * 100
        
        return {
            "consensus_score": round(consensus, 2),
            "burstiness": round(stylo["burstiness"], 3),
            "perplexity": round(ppl, 2),
            "confidence": "HIGH" if len(words) > 180 else "MODERATE"
        }
